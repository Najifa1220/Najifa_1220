// Arithmetic

let a = 30;
let b = 4;

console.log(a + b);
console.log(a - b);
console.log(a * b);
console.log(a / b);
console.log(a % b);
console.log(a ** b);

// Assignment

let x = 5;

x += 5;
x -= 2;
x *= 2;
x /= 2;

console.log(x);

// Comparison

console.log(10 > 5);
console.log(10 < 5);
console.log(10 >= 10);
console.log(10 === 10);
console.log(10 !== 5);

// Logical

let age = 22;
let hasID = true;

console.log(age >= 18 && hasID);
console.log(age < 18 || hasID);
console.log(!hasID);

// Ternary

let status = age >= 18 ? "Adult" : "Minor";
console.log(status);
