// Basic object

let student = {
  name: "Meherin",
  age: 22,
  department: "CSE",
};

console.log(student.name);
console.log(student.age);

// Bracket notation

console.log(student["name"]);

// Modifying an object

student.age = 22;
student.university = "Premier University";

console.log(student);
